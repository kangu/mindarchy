cmake_minimum_required(VERSION 3.21)
set(MINDMAP_VERSION "0.1.0" CACHE STRING "Application release version")
project(Mindarchy VERSION ${MINDMAP_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test Network)
add_library(mindmap_engine fonts.qrc src/engine.cpp src/engine.h src/theme.cpp src/theme.h src/drawing.cpp src/drawing.h src/preview.cpp src/preview.h)
target_link_libraries(mindmap_engine PUBLIC Qt6::Core Qt6::Gui)
target_include_directories(mindmap_engine PUBLIC src)
add_library(mindmap_canvas src/canvas.cpp src/canvasimages.cpp src/canvas.h src/shelltheme.cpp src/shelltheme.h)
target_link_libraries(mindmap_canvas PUBLIC mindmap_engine Qt6::Quick)
qt_add_executable(mindarchy src/main.cpp src/macapplication.cpp src/windowplacement.cpp src/windowplacement.h resources.qrc)
target_link_libraries(mindarchy PRIVATE mindmap_canvas Qt6::Qml Qt6::QuickControls2 Qt6::Network)
target_compile_definitions(mindarchy PRIVATE MINDMAP_VERSION="${PROJECT_VERSION}")
if(APPLE)
    enable_language(OBJCXX)
    find_package(Qt6 REQUIRED COMPONENTS Network)
    target_sources(mindarchy PRIVATE src/macwindow.mm src/macplacement.mm src/macapplication.h)
    target_link_libraries(mindarchy PRIVATE Qt6::Network)
    target_link_libraries(mindarchy PRIVATE "-framework AppKit" "-framework UniformTypeIdentifiers")
    add_executable(omm-preview packaging/macos/PreviewProvider.mm)
    set_target_properties(omm-preview PROPERTIES AUTOMOC OFF OUTPUT_NAME OMMPreview)
    target_compile_options(omm-preview PRIVATE -fobjc-arc)
    target_link_options(omm-preview PRIVATE -Wl,-e,_NSExtensionMain)
    target_link_libraries(omm-preview PRIVATE mindmap_engine "-framework Cocoa" "-framework Quartz" "-framework UniformTypeIdentifiers")
    set(preview_bundle "${CMAKE_CURRENT_BINARY_DIR}/mindarchy.app/Contents/PlugIns/OMMPreview.appex")
    set_target_properties(omm-preview PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${preview_bundle}/Contents/MacOS")
    configure_file(packaging/macos/Preview-Info.plist.in "${preview_bundle}/Contents/Info.plist" @ONLY)
    add_custom_command(TARGET omm-preview POST_BUILD
        COMMAND /usr/bin/codesign --force --sign - --entitlements "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Preview.entitlements" "${preview_bundle}")
    add_dependencies(mindarchy omm-preview)
    set(app_icon "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/mindarchy.icns")
    target_sources(mindarchy PRIVATE "${app_icon}")
    set_source_files_properties("${app_icon}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(mindarchy PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist.in"
        MACOSX_BUNDLE_ICON_FILE mindarchy.icns
        MACOSX_BUNDLE_GUI_IDENTIFIER org.mindarchy.app
        MACOSX_BUNDLE_BUNDLE_NAME "Mindarchy"
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
elseif(WIN32)
    enable_language(RC)
    string(REPLACE "." "," WINDOWS_VERSION "${PROJECT_VERSION}.0")
    set(APPLICATION_ICON_WINDOWS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/mindarchy.ico")
    configure_file(packaging/windows-icon.rc.in windows-icon.rc @ONLY)
    target_sources(mindarchy PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows-icon.rc" src/windowsdialogs.cpp)
    target_link_libraries(mindarchy PRIVATE comctl32 ole32 shell32)
    if(MSVC)
        target_compile_options(mindmap_engine PUBLIC /utf-8)
        target_link_options(mindarchy PRIVATE "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
    endif()
    install(TARGETS mindarchy RUNTIME DESTINATION .)
    set_target_properties(mindarchy PROPERTIES WIN32_EXECUTABLE TRUE)
else()
    include(GNUInstallDirs)
    install(TARGETS mindarchy RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES packaging/org.mindarchy.app.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES packaging/org.mindarchy.omm.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages)
    install(FILES packaging/org.mindarchy.app.thumbnailer DESTINATION ${CMAKE_INSTALL_DATADIR}/thumbnailers)
    foreach(size 16 24 32 48 64 128 256 512 1024)
        install(FILES assets/icons/mindarchy-${size}.png
            DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps
            RENAME org.mindarchy.app.png)
    endforeach()
endif()
enable_testing()
if(APPLE)
    qt_add_executable(macapplication_test tests/macapplication_test.mm src/macapplication.cpp src/macwindow.mm src/windowplacement.cpp src/macplacement.mm resources.qrc)
    target_link_libraries(macapplication_test PRIVATE mindmap_canvas Qt6::Qml Qt6::QuickControls2 Qt6::Network Qt6::Test "-framework AppKit" "-framework UniformTypeIdentifiers")
    add_test(NAME macapplication COMMAND macapplication_test)
    set_tests_properties(macapplication PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa" TIMEOUT 45)
    add_executable(macwindow_test tests/macwindow_test.mm src/macwindow.mm)
    target_link_libraries(macwindow_test PRIVATE Qt6::Gui "-framework AppKit" "-framework UniformTypeIdentifiers")
    add_test(NAME macwindow COMMAND macwindow_test)
    set_tests_properties(macwindow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa")
endif()

    target_sources(windowplacement_test PRIVATE tests/maczoom_test_support.mm src/macplacement.mm)

add_test(NAME preview COMMAND preview_test)

if(WIN32 AND NOT CMAKE_CROSSCOMPILING)
    # The offscreen QPA plugin needs an explicit font directory on Windows.
    set_property(TEST engine canvas ui windowplacement preview APPEND PROPERTY
        ENVIRONMENT "QT_QPA_FONTDIR=$ENV{WINDIR}/Fonts")
endif()

if(WIN32)
    qt_add_executable(windowsdialogs_test tests/windowsdialogs_test.cpp src/windowsdialogs.cpp)
    target_link_libraries(windowsdialogs_test PRIVATE mindmap_engine Qt6::Quick Qt6::Test comctl32 ole32 shell32)
    if(MSVC)
        target_link_options(windowsdialogs_test PRIVATE "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
    endif()
    add_test(NAME windowsdialogs COMMAND windowsdialogs_test)
    set_tests_properties(windowsdialogs PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows" TIMEOUT 45)

    qt_add_executable(application_test tests/application_test.cpp src/macapplication.cpp src/windowplacement.cpp resources.qrc)
    target_link_libraries(application_test PRIVATE mindmap_canvas Qt6::Qml Qt6::QuickControls2 Qt6::Network Qt6::Test)
    if(WIN32)
        target_sources(application_test PRIVATE src/windowsdialogs.cpp)
        target_link_libraries(application_test PRIVATE comctl32 ole32 shell32)
    endif()
    add_test(NAME application COMMAND application_test)

target_link_libraries(manualplacement_test PRIVATE mindmap_engine Qt6::Test)
